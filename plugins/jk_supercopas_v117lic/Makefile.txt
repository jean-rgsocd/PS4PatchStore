TOOLCHAIN := $(OO_PS4_TOOLCHAIN)
SDK       := ../goldhen_sdk

CFLAGS    := --target=x86_64-pc-freebsd12-elf -fPIC -funwind-tables -O1 -c -isysroot $(TOOLCHAIN) -isystem $(TOOLCHAIN)/include -I$(SDK)/include -Isource
LDFLAGS   := -m elf_x86_64 -pie --export-dynamic --script $(TOOLCHAIN)/link.x -e _init --eh-frame-hdr -L$(TOOLCHAIN)/lib -L$(SDK)

	clang $(CFLAGS) -o $@ $<

	ld.lld $(OBJS) $(SDK)/build/crtprx.o -o build/jk_supercopas_v117.elf $(LDFLAGS) $(LIBS)
	$(TOOLCHAIN)/bin/linux/create-fself -in=build/jk_supercopas_v117.elf -out=build/jk_supercopas_v117.oelf --lib=$(TARGET) --paid 0x3800000000000011

clean:
	rm -rf build $(TARGET)
